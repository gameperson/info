import numpy as np

class DervishDanceEngine:
    def __init__(self, baseline_vector, torsion_limit=1.5):
        self.p0 = np.array(baseline_vector, dtype=float)
        self.torsion_limit = torsion_limit
        self.current_state = self.p0.copy()
        self.trajectory_log = [self.p0.copy()]

    def induce_torsional_drift(self, perturbation_scale=0.5):
        """Simulates centrifugal release: drifting away from baseline into the Torsion Space."""
        noise = np.random.normal(0, perturbation_scale, size=self.p0.shape)
        self.current_state = self.current_state + noise
        self.trajectory_log.append(self.current_state.copy())
        return self.current_state

    def evaluate_torsion(self):
        """Measures angular and Euclidean displacement from the original purpose."""
        displacement = np.linalg.norm(self.current_state - self.p0)
        return displacement

    def execute_vulnerability_cutoff(self, candidate_pool):
        """
        Screening cutoff: Instead of deep renormalization back to origin,
        evaluates the drifted outliers and snaps to the one maximizing utility 
        while remaining bound to the core telos.
        """
        best_score = -float('inf')
        optimal_state = self.p0

            # Calculate utility proxy (e.g., magnitude/novelty)
            utility = np.linalg.norm(candidate)
            # Calculate alignment cosine similarity with original purpose

            # Score balances utility against teleological retention
            score = utility * max(0, cos_sim)

            if score > best_score:
                best_score = score
                optimal_state = candidate

        # Re-anchor baseline to the deviant outlier
        self.p0 = optimal_state
        self.current_state = optimal_state.copy()
        return self.p0
